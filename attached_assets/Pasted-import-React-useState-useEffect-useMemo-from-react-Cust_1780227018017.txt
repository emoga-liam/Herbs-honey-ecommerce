export default function App() {
  // --- NAVIGATION STATE ---
  const [currentTab, setCurrentTab] = useState('home'); // 'home' | 'shop' | 'about' | 'contact' | 'checkout' | 'success'
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
  
  // --- STORE STATE ---
  const [cart, setCart] = useState([]);
  const [cartOpen, setCartOpen] = useState(false);
  const [toast, setToast] = useState(null);
  const [badgeAnimation, setBadgeAnimation] = useState(false);
  
  // Flagship Product Quick Buy states on Home Page
  const [flagshipSize, setFlagshipSize] = useState("500g");
  const [flagshipQty, setFlagshipQty] = useState(1);
  
  // Catalog / Search states
  const [catalogSearch, setCatalogSearch] = useState('');
  const [catalogCategory, setCatalogCategory] = useState('all');
  
  // Modal detail view
  const [viewProduct, setViewProduct] = useState(null);
  const [viewProductSize, setViewProductSize] = useState("");

  // Promo Code States
  const [promoCode, setPromoCode] = useState('');
  const [discountPercent, setDiscountPercent] = useState(0); // e.g. 0.20 for 20%

  // Checkout Form States (Localized for Nigerian shipping flow)
  const [checkoutForm, setCheckoutForm] = useState({
    name: '',
    email: '',
    address: '',
    city: '',
    stateRegion: 'Lagos',
    phone: '',
    cardNumber: '',
    expiry: '',
    cvv: ''
  });
  const [lastOrder, setLastOrder] = useState(null);

  // Contact Page Form States
  const [contactForm, setContactForm] = useState({
    name: '',
    email: '',
    subject: '',
    message: ''
  });

  // Auto-scroll to top on tab change
  useEffect(() => {
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }, [currentTab]);

  // Alert/Toast controller
  const triggerToast = (msg) => {
    setToast(msg);
    setTimeout(() => setToast(null), 3000);
  };

  // --- CART HANDLERS ---
  const addToCart = (product, size, qty = 1) => {
    setCart(prev => {
      const matchIndex = prev.findIndex(item => item.product.id === product.id && item.size === size);
      if (matchIndex > -1) {
        const next = [...prev];
        next[matchIndex].quantity += qty;
        return next;
      }
      return [...prev, { product, size, quantity: qty }];
    });
    
    // Trigger intense visual pulsing feedback on the basket icon
    setBadgeAnimation(true);
    setTimeout(() => setBadgeAnimation(false), 500);
    
    // Explicitly do not open the cart drawer automatically. Only show the success toast.
    triggerToast(`Added ${qty}x ${product.title} (${size}) to your cart!`);
  };

  const updateCartQty = (productId, size, change) => {
    setCart(prev => {
      return prev.map(item => {
        if (item.product.id === productId && item.size === size) {
          const nextQty = item.quantity + change;
          return nextQty > 0 ? { ...item, quantity: nextQty } : null;
        }
        return item;
      }).filter(Boolean);
    });
  };

  const removeItem = (productId, size) => {
    setCart(prev => prev.filter(item => !(item.product.id === productId && item.size === size)));
    triggerToast("Item removed from your cart.");
  };

  // Pricing calculations (Naira Based ₦)
  const cartSubtotal = useMemo(() => {
    return cart.reduce((sum, item) => sum + (item.product.price * item.quantity), 0);
  }, [cart]);

  const discountAmount = useMemo(() => {
    return cartSubtotal * discountPercent;
  }, [cartSubtotal, discountPercent]);

  // Delivery fee: Free for orders over ₦45,000, otherwise flat ₦3,500 for dispatch
  const deliveryFee = useMemo(() => {
    return cartSubtotal > 45000 ? 0 : (cartSubtotal > 0 ? 3500 : 0);
  }, [cartSubtotal]);

  const cartTotal = useMemo(() => {
    return Math.max(0, cartSubtotal - discountAmount + deliveryFee);
  }, [cartSubtotal, discountAmount, deliveryFee]);

  // Handle promo code submission
  const handleApplyPromo = (e) => {
    e.preventDefault();
    if (promoCode.trim().toUpperCase() === 'GOLDEN20') {
      setDiscountPercent(0.20);
      triggerToast("Promo 'GOLDEN20' applied! 20% off your subtotal.");
    } else {
      triggerToast("Invalid discount coupon.");
    }
  };

  // Filter products for the shop view
  const filteredProducts = useMemo(() => {
    return PRODUCTS.filter(p => {
      const matchesCat = catalogCategory === 'all' || p.category === catalogCategory;
      const matchesSearch = p.title.toLowerCase().includes(catalogSearch.toLowerCase()) || 
                            p.desc.toLowerCase().includes(catalogSearch.toLowerCase());
      return matchesCat && matchesSearch;
    });
  }, [catalogCategory, catalogSearch]);

  // Form handle changes
  const handleInputChange = (e) => {
    setCheckoutForm({ ...checkoutForm, [e.target.name]: e.target.value });
  };

  const handleContactChange = (e) => {
    setContactForm({ ...contactForm, [e.target.name]: e.target.value });
  };

  // Submit Contact Form
  const handleContactSubmit = (e) => {
    e.preventDefault();
    triggerToast(`Message sent! Thank you, ${contactForm.name}. Our apiculturists will reply soon.`);
    setContactForm({ name: '', email: '', subject: '', message: '' });
  };

  // Submit Order (Familiar payment processing simulation with Paystack/Flutterwave design references)
  const handlePlaceOrder = (e) => {
    e.preventDefault();
    if (!checkoutForm.name || !checkoutForm.email || !checkoutForm.address || !checkoutForm.phone) {
      triggerToast("Please complete all required shipping fields.");
      return;
    }

    const receiptNum = "NEX-NG-" + Math.floor(100000 + Math.random() * 900000);
    setLastOrder({
      id: receiptNum,
      customer: checkoutForm,
      items: [...cart],
      subtotal: cartSubtotal,
      discount: discountAmount,
      delivery: deliveryFee,
      total: cartTotal
    });

    // Reset store state
    setCart([]);
    setDiscountPercent(0);
    setPromoCode('');
    setCheckoutForm({
      name: '',
      email: '',
      address: '',
      city: '',
      stateRegion: 'Lagos',
      phone: '',
      cardNumber: '',
      expiry: '',
      cvv: ''
    });
    
    setCurrentTab('success');
  };

  // Robust function to navigate and smoothly scroll to flagship product section
  const handleShopFlagshipScroll = () => {
    setCurrentTab('home');
    setMobileMenuOpen(false);
    setTimeout(() => {
      const targetElement = document.getElementById('flagship-buy-now');
      if (targetElement) {
        const offset = 100; // navbar buffer offset
        const bodyRect = document.body.getBoundingClientRect().top;
        const elementRect = targetElement.getBoundingClientRect().top;
        const elementPosition = elementRect - bodyRect;
        const offsetPosition = elementPosition - offset;

        window.scrollTo({
          top: offsetPosition,
          behavior: 'smooth'
        });
      }
    }, 150);
  };

  return (
    <div className="bg-[#fbfbfa] text-stone-900 font-sans min-h-screen relative antialiased flex flex-col justify-between">
      
      {/* Dynamic Gradients for Custom SVG jar assets */}
      <svg className="absolute w-0 h-0" xmlns="http://www.w3.org/2000/svg">
        <defs>
          <linearGradient id="honeyGrad" x1="0%" y1="0%" x2="100%" y2="100%">
            <stop offset="0%" stopColor="#f7e491" />
            <stop offset="40%" stopColor="#e5b128" />
            <stop offset="100%" stopColor="#855619" />
          </linearGradient>
          <linearGradient id="royalGrad" x1="0%" y1="0%" x2="100%" y2="100%">
            <stop offset="0%" stopColor="#ffffff" />
            <stop offset="50%" stopColor="#fdfbeb" />
            <stop offset="100%" stopColor="#f7e491" />
          </linearGradient>
        </defs>
      </svg>

      {/* --- FLOATING TOAST --- */}
      {toast && (
        <div className="fixed bottom-6 right-6 z-50 bg-[#101d18]/95 backdrop-blur border border-amber-400 px-6 py-4 rounded-xl shadow-xl flex items-center gap-3 animate-slide-in max-w-sm">
          <span className="w-2 h-2 rounded-full bg-amber-400 animate-ping"></span>
          <span className="text-xs font-semibold tracking-wide">{toast}</span>
        </div>
      )}

      {/* --- TOP HEADER NAVIGATION --- */}
      <header className="sticky top-0 left-0 w-full z-40 bg-[#101d18] text-white border-b border-white/10 shadow-md">
        <div className="max-w-7xl mx-auto px-6 h-20 flex items-center justify-between">
          
          {/* Logo */}
          <button onClick={() => { setCurrentTab('home'); setMobileMenuOpen(false); }} className="flex items-center gap-3 text-left group">
            <div className="w-10 h-10 rounded-xl bg-amber-400 flex items-center justify-center text-stone-950 font-bold shadow-md transform group-hover:rotate-6 transition-transform">
              🍯
            </div>
            <div>
              <span className="text-lg font-black tracking-tight block">Nectar<span className="text-amber-400">Elix</span></span>
              <span className="text-[9px] uppercase tracking-widest text-stone-400 block -mt-1 font-bold">Raw Nigerian Honey</span>
            </div>
          </button>

          {/* Familiar Center Navigation Menu (Routes perfectly between pages on desktop) */}
          <nav className="hidden lg:flex items-center gap-8">
            <button onClick={() => setCurrentTab('home')} className={`text-xs font-black uppercase tracking-widest transition-colors ${currentTab === 'home' ? 'text-amber-400' : 'text-stone-300 hover:text-amber-400'}`}>
              Home
            </button>
            <button onClick={() => { setCatalogCategory('all'); setCurrentTab('shop'); }} className={`text-xs font-black uppercase tracking-widest transition-colors ${currentTab === 'shop' ? 'text-amber-400' : 'text-stone-300 hover:text-amber-400'}`}>
              Shop Range
            </button>
            <button onClick={() => setCurrentTab('about')} className={`text-xs font-black uppercase tracking-widest transition-colors ${currentTab === 'about' ? 'text-amber-400' : 'text-stone-300 hover:text-amber-400'}`}>
              Our Story
            </button>
            <button onClick={() => setCurrentTab('contact')} className={`text-xs font-black uppercase tracking-widest transition-colors ${currentTab === 'contact' ? 'text-amber-400' : 'text-stone-300 hover:text-amber-400'}`}>
              Contact Us
            </button>
          </nav>

          {/* Search Shortcut, Cart Widget, and Mobile Hamburger trigger */}
          <div className="flex items-center gap-4">
            {/* Quick Search trigger which opens Shop Tab */}
            <button onClick={() => { setCurrentTab('shop'); setMobileMenuOpen(false); }} className="p-2 text-stone-300 hover:text-amber-400 transition-colors" title="Search Range">
              <svg className="w-5 h-5" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z"></path></svg>
            </button>
            
            {/* Real-time Shopping Cart Button */}
            <button onClick={() => { setCartOpen(true); setMobileMenuOpen(false); }} className="relative p-2.5 bg-white/5 hover:bg-white/10 border border-white/10 rounded-xl transition-all" title="View Reservation Cart">
              <svg className="w-5 h-5 text-amber-400" fill="none" stroke="currentColor" strokeWidth="2.5" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" d="M16 11V7a4 4 0 00-8 0v4M5 9h14l1 12H4L5 9z"></path></svg>
              {cart.length > 0 && (
                <span className={`absolute -top-1.5 -right-1.5 w-5 h-5 bg-amber-500 text-stone-950 font-black text-[10px] rounded-full flex items-center justify-center shadow-md ${badgeAnimation ? 'animate-cart-pulse' : ''}`}>
                  {cart.reduce((sum, item) => sum + item.quantity, 0)}
                </span>
              )}
            </button>

            {/* Mobile Hamburger menu button */}
            <button 
              onClick={() => setMobileMenuOpen(!mobileMenuOpen)} 
              className="lg:hidden p-2 text-stone-300 hover:text-white transition-colors"
              aria-label="Toggle Menu"
            >
              {mobileMenuOpen ? (
                <svg className="w-6 h-6 text-amber-400" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12"/>
                </svg>
              ) : (
                <svg className="w-6 h-6" fill="none" stroke="currentColor" strokeWidth="2" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M4 6h16M4 12h16m-7 6h7"/>
                </svg>
              )}
            </button>
          </div>
        </div>

        {/* --- MOBILE DRAW-DOWN DRAWER (Mobile First) --- */}
        <div className={`lg:hidden transition-all duration-350 ease-in-out border-t border-white/10 ${mobileMenuOpen ? 'max-h-72 py-4' : 'max-h-0 py-0 overflow-hidden'}`}>
          <div className="flex flex-col px-6 gap-4 text-left">
            <button onClick={() => { setCurrentTab('home'); setMobileMenuOpen(false); }} className={`py-2 text-sm font-black uppercase tracking-widest text-left ${currentTab === 'home' ? 'text-amber-400' : 'text-stone-300'}`}>
              Home
            </button>
            <button onClick={() => { setCatalogCategory('all'); setCurrentTab('shop'); setMobileMenuOpen(false); }} className={`py-2 text-sm font-black uppercase tracking-widest text-left ${currentTab === 'shop' ? 'text-amber-400' : 'text-stone-300'}`}>
              Shop Range
            </button>
            <button onClick={() => { setCurrentTab('about'); setMobileMenuOpen(false); }} className={`py-2 text-sm font-black uppercase tracking-widest text-left ${currentTab === 'about' ? 'text-amber-400' : 'text-stone-300'}`}>
              Our Story
            </button>
            <button onClick={() => { setCurrentTab('contact'); setMobileMenuOpen(false); }} className={`py-2 text-sm font-black uppercase tracking-widest text-left ${currentTab === 'contact' ? 'text-amber-400' : 'text-stone-300'}`}>
              Contact Us
            </button>
          </div>
        </div>
      </header>

      {/* --- PERSISTENT SUB-HERO ANNOUNCEMENT BAR --- */}
      <div className="bg-amber-400 text-stone-950 py-2.5 text-center text-[11px] font-black uppercase tracking-widest shadow-inner px-4">
        🎉 Free Delivery Nationwide On All Orders Over ₦45,000! Use Code <span className="underline font-bold">GOLDEN20</span> for 20% off your purchase!
      </div>

      {/* --- PRIMARY VIEWS SWITCHER --- */}
      <main className="flex-grow">
        
        {/* ========================================================= */}
        {/* 1. HOMEPAGE VIEW (Familiar Scrollable Landing + Shop combo) */}
        {/* ========================================================= */}
        {currentTab === 'home' && (
          <div>
            {/* HERO SECTION */}
            <section className="relative bg-[#101d18] text-white py-20 px-6 overflow-hidden">
              <div className="absolute inset-0 bg-[radial-gradient(circle_at_70%_30%,rgba(229,177,40,0.1),transparent_50%)]"></div>
              
              <div className="max-w-7xl mx-auto grid grid-cols-1 lg:grid-cols-12 gap-12 items-center relative z-10">
                {/* Left Text details */}
                <div className="lg:col-span-7 space-y-6 text-left">
                  <div className="inline-flex items-center gap-2 px-3 py-1 bg-white/5 border border-white/10 rounded-full text-[10px] font-black uppercase tracking-widest text-amber-400">
